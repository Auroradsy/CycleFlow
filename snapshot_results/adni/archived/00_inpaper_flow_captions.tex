\begin{figure}[t]
\centering
\includegraphics[width=0.92\linewidth]{adni/00_inpaper_flow_features_a2b_morph.pdf}
\caption{Feature maps along the flow, T1$\rightarrow$FA (\emph{morph}), for five
held-out subjects; each subject occupies two rows. \emph{PCA}: the
$256\times28\times28$ state after every block $f_k$, projected onto three
principal directions and shown as RGB, between the source encoding
$E_{\mathrm{T1}}(x)$ and the encoding $E_{\mathrm{FA}}(y)$ of the paired FA
image. The basis is fitted once over all panels, so a colour denotes the same
direction of the 256-dimensional channel space everywhere in the figure --- only
$3$ of those $256$ directions are visible, and the colours carry no meaning
across figures. \emph{$\|\Delta\|$}: the per-location RMS change from the
previous column, which sees all $256$ channels; under $E_{\mathrm{FA}}(y)$ it is
the gap that remains between the flow output and that encoding. The first two
blocks act mainly on the brain boundary and around the ventricles, while the last
two rewrite the whole parenchyma and bring the colouring close to
$E_{\mathrm{FA}}(y)$. Over the test set the relative distance to
$E_{\mathrm{FA}}(y)$ falls monotonically across the blocks
($1.29\rightarrow1.20\rightarrow1.08\rightarrow0.95\rightarrow0.82$): the flow
moves the code towards the target encoder's representation without landing on
it.}
\label{fig:adni-flow-features}
\end{figure}

\begin{figure}[t]
\centering
\includegraphics[width=\linewidth]{adni/00_inpaper_flow_affine_a2b_morph.pdf}
\caption{Affine terms of the coupling layers, T1$\rightarrow$FA, for the subjects
of Figure~\ref{fig:adni-flow-features}. Each coupling splits the $256$ channels
in half and updates one half as $x_2\mapsto a\odot x_2+b$, with
$a=\exp(\tanh s(x_1))$ and $b=t(x_1)$ predicted from the other half, which passes
through unchanged; that split is what makes the block invertible in closed form,
so the panels show the update applied to the $128$ channels that change. For each
block, the left panel is $\log a$ averaged over those channels (red expands, blue
contracts), on a logarithmic scale because $a$ acts multiplicatively, is
symmetric about $1$, and $\log a$ is both the bounded network output $\tanh s$
and the per-location log-determinant; the right panel is the RMS of $b$ over the
same channels, since the channel mean of $b$ is about a tenth of its RMS --- the
shifts largely cancel across channels, leaving the magnitude as what is
informative. Blocks $1$ and $2$ expand the interior and contract the rim; blocks
$3$ and $4$ contract almost everywhere and apply the largest shifts, strongest
around the ventricles. Locations outside the brain are white.}
\label{fig:adni-flow-affine}
\end{figure}

\begin{figure}[t]
\centering
\includegraphics[width=\linewidth]{adni/00_inpaper_cycle_a2b_morph.pdf}
\caption{Translating T1$\rightarrow$FA and back (\emph{morph}, five held-out
slices). \emph{Translated} is $D_{\mathrm{FA}}(f(E_{\mathrm{T1}}(x)))$; the two
cycles differ in where the return trip starts. The \emph{latent cycle} inverts
the flow output directly, $D_{\mathrm{T1}}(f^{-1}(f(E_{\mathrm{T1}}(x))))$:
since $f^{-1}\!\circ\!f$ is the identity by construction (at most
$7.6\times10^{-6}$ in the latent), this is plain autoencoding and its error
measures the encoder/decoder alone (SSIM $0.998$). The \emph{image cycle}
decodes the translation to pixels and re-encodes it,
$D_{\mathrm{T1}}(f^{-1}(E_{\mathrm{FA}}(\hat y)))$, adding the round trip
through image space and any mismatch with the latent the encoder assigns to it
(SSIM $0.984$, MAE $0.010$). Both error columns share one scale, so the gap
between them --- on the cortical rim and fine white-matter detail --- is the
price of re-encoding, not of the flow.}
\label{fig:adni-cycle-a2b}
\end{figure}

\begin{figure}[t]
\centering
\includegraphics[width=\linewidth]{adni/00_inpaper_cycle_b2a_morph.pdf}
\caption{The same cycles in the FA$\rightarrow$T1 direction, with the layout of
Figure~\ref{fig:adni-cycle-a2b}. The latent cycle again reproduces autoencoding
(latent error at most $9.9\times10^{-6}$, SSIM $0.998$), while the image cycle
reaches SSIM $0.980$ and mean absolute error $0.011$, with the residual on fine,
high-anisotropy detail and largest for the noisiest FA input (fourth row).}
\label{fig:adni-cycle-b2a}
\end{figure}

\begin{figure}[h]

\caption{Feature maps along the flow, FA$\rightarrow$T1, for the subjects of
Figure~\ref{fig:adni-flow-features}, with the same layout and the same PCA basis.
The inverse blocks run in reverse order, $f_4^{-1}$ first. The colouring moves
from that of $E_{\mathrm{FA}}(x)$ to that of $E_{\mathrm{T1}}(y)$. The largest
change comes from $f_3^{-1}$ around the ventricles, while $f_2^{-1}$ and
$f_1^{-1}$ act mostly on the rim and the ventricles, as their forward
counterparts do. Over the test set the relative distance to $E_{\mathrm{T1}}(y)$
falls from $1.29$ to $1.16$, $1.05$, $0.94$ and $0.79$.}
\label{fig:adni-flow-features-b2a}
\end{figure}

\begin{figure}[h]
\centering
\includegraphics[width=\linewidth]{adni/00_inpaper_flow_affine_b2a_morph.pdf}
\caption{Affine terms applied in the FA$\rightarrow$T1 direction, with the layout
of Figure~\ref{fig:adni-flow-affine}. Travelling backwards, each coupling applies
the inverse update $x_2=a'\odot y_2+b'$ with $a'=1/a$ and $b'=-b/a$, and these
are the quantities shown. $f_2^{-1}$ and $f_1^{-1}$ contract the interior and
expand the rim, the mirror image of blocks $1$ and $2$ in the forward direction,
and they apply the largest shifts.}
\label{fig:adni-flow-affine-b2a}
\end{figure}

\begin{figure}[h]

\caption{Learned ActNorm parameters of the four blocks. Each ActNorm applies
$x\mapsto ax+b$ with one $a$ and one $b$ per channel. There is one dot per
channel ($256$ per block) and the bar marks the median. All values stay within
$|\log a|\le0.1$ and $|b|<0.05$, so ActNorm remains close to the identity and the
translation is carried by the input-dependent couplings. The inverse direction
applies $1/a$ and $-b/a$.}
